# WSL GPU Sharing 명령어 모음
# 위치: /mnt/c/Users/장성욱/Desktop/ws/GPU-sharing-Jang

# ============================================
# 1. 전체 실행 (처음 또는 전체 재빌드)
# ============================================
docker-compose up --build -d

# ============================================
# 2. 코드 수정 후 재빌드
# ============================================

# 프론트엔드만 수정했을 때
docker-compose up --build -d frontend

# 백엔드만 수정했을 때  
docker-compose up --build -d backend

# 둘 다 수정했을 때
docker-compose up --build -d

# ============================================
# 3. 빠른 재시작 (빌드 없이)
# ============================================
docker-compose restart frontend
docker-compose restart backend
docker-compose restart  # 둘 다

# ============================================
# 4. 로그 확인
# ============================================
docker-compose logs -f                    # 전체 로그
docker-compose logs -f frontend          # 프론트엔드만
docker-compose logs -f backend           # 백엔드만

# ============================================
# 5. 중지/삭제
# ============================================
docker-compose down                       # 중지 (데이터 유지)
docker-compose down -v                    # 중지 + 데이터베이스 삭제
docker-compose down --rmi all -v          # 완전 초기화 (이미지도 삭제)

# ============================================
# 6. 상태 확인
# ============================================
docker-compose ps                         # 실행 중인 서비스
docker ps                                 # 전체 컨테이너

# ============================================
# 7. 단일 컨테이너 명령어
# ============================================
docker logs -f gpu-sharing-frontend       # 프론트 로그
docker logs -f gpu-sharing-backend        # 백엔드 로그
docker exec -it gpu-sharing-backend sh    # 백엔드 내부 접속
docker exec -it gpu-sharing-frontend sh   # 프론트엔드 내부 접속

# ============================================
# 8. 캐시 삭제 후 완전 재빌드 (문제 해결용)
# ============================================
docker system prune -f                    # 사용 안 하는 이미지/컨테이너 정리
docker-compose build --no-cache           # 캐시 없이 빌드
docker-compose up -d

# ============================================
# 9. 네트워크 문제 해결
# ============================================
docker network ls                         # 네트워크 목록
docker network rm gpu-sharing-jang_gpu-sharing-network  # 네트워크 삭제
docker-compose up -d                      # 재생성
